import itertools import string import time # كلمة السر التجريبية target = "abc7" # الأحرف المسموح بتجربتها characters = string.ascii_lowercase + string.digits attempts = 0 start = time.time() for length in range(1, len(target) + 1): for guess_tuple in itertools.product(characters, repeat=length): guess = ''.join(guess_tuple) attempts += 1 print(f"\rمحاولة رقم {attempts}: {guess}", end="") if guess == target: elapsed = time.time() - start print("\n\nتم العثور على كلمة السر!") print(f"كلمة السر: {guess}") print(f"عدد المحاولات: {attempts}") print(f"الوقت: {elapsed:.4f} ثانية") raise SystemExit print("\nلم يتم العثور على كلمة السر.")
